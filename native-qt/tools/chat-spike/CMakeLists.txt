cmake_minimum_required(VERSION 3.24)
project(chat_spike LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "chat-spike requires Windows/WebView2")
endif()
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
set(WEBVIEW2 "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/webview2")
qt_add_executable(chat-spike main.cpp)
qt_add_resources(chat-spike scripts PREFIX / FILES chrome-shim.js bootstrap.js)
target_include_directories(chat-spike PRIVATE "${WEBVIEW2}/build/native/include")
target_compile_definitions(chat-spike PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_link_libraries(chat-spike PRIVATE Qt6::Widgets "${WEBVIEW2}/build/native/x64/WebView2LoaderStatic.lib" ole32 oleaut32 user32 version shlwapi psapi)
set_target_properties(chat-spike PROPERTIES WIN32_EXECUTABLE FALSE)
